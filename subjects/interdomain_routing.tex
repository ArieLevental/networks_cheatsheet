\section{Interdomain Routing \& BGP}

\subsection{Autonomous Systems (AS)}
\begin{itemize}
    \item \textbf{Def.:} IP networks/routers under one admin with a common routing policy.
    \item \textbf{ASN:} is a unique 16/32-bit AS identifier used in BGP.
    \item \textbf{AS relationships:}
    \begin{itemize}
        \item \textbf{Customer / Provider:} Customer pays provider for transit.
        \item \textbf{Full / Partial transit:} Full = global reachability; partial = limited reach.
        \item \textbf{Peering:} Settlement-free; advertise own + customers only.
    \end{itemize}
    \item \textbf{AS-level topology:} Common categories
    \begin{itemize}
        \item \textbf{Tier-1} ($\sim 10$ ASes, e.g. AT\&T): No providers; global reach via peering.
        \item \textbf{Tier-2:} Providers + peers (buys some transit).
        \item \textbf{Stub AS} ($85\%$ of ASes): Single provider; no downstream customers.
    \end{itemize}
\end{itemize}

\subsection{Border Gateway Protocol (BGP)}
\begin{itemize}
    \item \textbf{Definition:} Inter-AS routing protocol (Policy-based).
    \item \textbf{Path-vector:} Advertises AS\_PATH loops avoided by rejecting own ASN.
    \item \textbf{Policy + attributes:} LOCAL\_PREF, MED, AS\_PATH, NEXT\_HOP, etc.
    \item \textbf{Priority Hierarchy:}
    \begin{itemize}
        \item \textbf{1. Forwarding Plane (LPM):} When forwarding a packet, the router \textit{always} chooses the \textbf{most specific prefix} (Longest Prefix Match).
        \item \textbf{2. Control Plane (BGP Attributes):} BGP attributes (LocalPref, AS\_PATH) are \textit{only} used to choose the best path between multiple routes to the \textbf{exact same prefix}.
        \item \textit{Example:} A path to \texttt{1.0.1.0/24} (via long path) ALWAYS beats \texttt{1.0.0.0/16} (via short path), because /24 is more specific. Attributes are ignored here.
    \end{itemize}
\end{itemize}

\subsection{BGP message flow (5 stages)}
\begin{enumerate}
    \item \textbf{TCP setup:} Session on TCP/179.
    \item \textbf{OPEN:} Exchange ASN, hold-time, capabilities.
    \item \textbf{Initial UPDATE:} Send NLRI + attributes.
    \item \textbf{Decision:} Import policy + best-path selection.
    \item \textbf{Steady state:} KEEPALIVE + incremental UPDATE/NOTIFICATION.
\end{enumerate}


\subsection{iBGP, eBGP and IGP interaction}
\begin{itemize}
    \item \textbf{eBGP:} Between ASes; increments AS\_PATH, may change NEXT\_HOP.
    \item \textbf{iBGP:} Within AS; distributes reachability to \textit{external} prefixes and maps them to an egress point.
    \item \textbf{IGP:} Computes shortest paths \textit{within} the AS and provides reachability to BGP NEXT\_HOP / egress routers.
    \item \textbf{Best-path inside AS (hot-potato):} \textcolor{gray}{(1)} Highest LOCAL\_PREF, then \textcolor{gray}{(2)} shortest AS\_PATH, then \textcolor{gray}{(3)} closest egress (lowest IGP cost), then \textcolor{gray}{(4)} tie-break.
\end{itemize}

\subsection{BGP Stability}
\begin{itemize}
    \item \textbf{Stable state:} Routes converge with no further changes.
    \item \textbf{Safety theorem (informal):}
    \begin{itemize}
		\item \textbf{Theorem:} The existence of more than one stable BGP state in a network implies that it is \textbf{not} BGP safe.
        \item Under Gao-Rexford, BGP converges to a (single) stable outcome.
    \end{itemize}
    \subsubsection{Gao-Rexford conditions (commercial routing)}
    \begin{itemize}
        \item \textbf{Topology:}  No customer-provider cycles in the AS graph.
        \item \textbf{Preference:} Prefer routes from customers over peers/providers.
        \item \textbf{Export:} Advertise customer routes to all; peer/provider routes only to customers.
    \end{itemize}
\end{itemize}

\subsection{BGP Security}
\begin{itemize}
    \item \textbf{Trust-based, weak by default:} BGP updates are accepted unless filtered; vulnerable to mistakes and attacks.
    \item \textbf{Session security (TCP-based):} eavesdropping, spoofing, reset, or tampering with UPDATEs.
    \item \textbf{Prefix hijacking:} False origin claims or more-specific prefixes attract traffic (can be accidental or malicious).
    \item \textbf{Detection:} Monitor multiple vantage points, anomaly alerts, route collectors/looking glasses.
    \item \textbf{Origin authentication:} Using a secure database to validate which AS may announce a prefix (reduces hijacks but not full path validation).
\end{itemize}
