\section{Link Layer \& Local Area Networks (LAN)}
\subsection{Shared Medium Intuition}
\begin{itemize}
	\item Nodes broadcast on same channel; only one can transmit at a time.
	\item Simultaneous transmit $\Rightarrow$ collision $\Rightarrow$ data lost.
	\item Sender uses full link bandwidth while transmitting.
\end{itemize}

\subsection{Definitions}
\begin{itemize}
	\item \textbf{Bandwidth:} max data rate of channel ($B$).
	\item \textbf{Throughput:} fraction of bandwidth used to send traffic.
	\item \textbf{Goodput:} fraction of bandwidth used for successful data.
	\item $\eta=\frac{\text{avg effective bandwidth}}{\text{total bandwidth}}=\frac{\text{avg effective time}}{\text{total time}}$, $0\le\eta\le1$.
\end{itemize}

\subsection{Multiple Access Domains}
\begin{itemize}
    \item \textbf{Multiple Access Domain:} set of nodes sharing a common channel.
    \item \textbf{Channel Partitioning:} divide channel into smaller "pieces" (time/freq/code).
    \item \textbf{Random Access:} transmit at will; handle collisions (ALOHA, CSMA).
    \item \textbf{Taking Turns:} nodes take turns to transmit (polling/token passing).
\end{itemize}
\subsection{ALOHA protocol}
\begin{itemize}
	\item Random access on shared medium; collisions if overlap. Frame time $T$ (size $L$, rate $B$): $T=L/B$.
    \item \textbf{Binomial model:} $N$ users, each transmits in a frame time $T$ w.p. $p$.
    \item \textbf{Poisson model:} Aggregate attempts are Poisson with rate $G$ (attempts per frame time $T$).
    \item \textbf{Slotted ALOHA} (align to slots of size $T$):
	\begin{itemize}
		\item Success if exactly one transmission in a slot.
        \item Binomial: $\mathbb{P}_{\text{suc}}=Np(1-p)^{N-1}$, max at $p^\star=1/N$ gives max goodput (when $N$ is large) of $\eta\approx 1/e$.
        \item Poisson: $\eta=Ge^{-G}$, Max $\eta_{\max}=1/e$ at $G=1$.
	\end{itemize}
	\item \textbf{Pure ALOHA} (unslotted):
	\begin{itemize}
		\item Vulnerable period $2T$ (current frame $\pm$ overlap).
        \item Binomial (heuristic): $\mathbb{P}_{\text{suc}} \approx Np(1-p)^{2(N-1)}$, max at $p^\star=1/(2N-1)$ gives max goodput (when $N$ is large) of $\eta\approx 1/(2e)$.
        \item Poisson: $\eta=Ge^{-2G}$. Max $\eta_{\max}=1/(2e)$ at $G=0.5$.
	\end{itemize}
\end{itemize}

\subsection{CSMA (Carrier Sense Multiple Access)}
\begin{itemize}
    \item \textbf{Basic Principle:} Listen (carrier sense) before transmitting.
    \item \textbf{Persistence Strategies:}
    \begin{itemize}
        \item \textbf{1-persistent:} If idle, transmit immediately. If busy, wait until idle and transmit immediately (high collision probability).
        \item \textbf{Non-persistent:} If idle, transmit. If busy, wait a random time (backoff) and check again.
        \item \textbf{p-persistent:} If idle, transmit with probability $p$, else wait for next slot (compromise).
    \end{itemize}
    \item \textbf{Propagation delay} $\tau$ ($T_{\text{prop}}$): Time for signal to travel between the two farthest nodes.
\end{itemize}

\subsubsection{CSMA/CD (Collision Detection) - Wired (Ethernet)}
\begin{itemize}
    \item \textbf{Mechanism:} Listen while transmitting. If collision detected (voltage spike), abort transmission, send \textbf{jam signal}, then backoff.
    \item \textbf{Condition for Detection:} Transmission time must be long enough to detect a collision from the farthest node and receive the collision signal back.
    \[ T_{\text{trans}} \ge 2 \cdot T_{\text{prop}} \]
    (This determines the minimum packet size).
	\item \textbf{Example (minimum frame size):} distance $=8\,\mathrm{km}$, propagation speed $=4\times 10^7\,\mathrm{m/s}$, rate $=5\,\mathrm{Mbps}$. Then
	\[ T_{\text{prop}}=\frac{8\times 10^3}{4\times 10^7}=2\times 10^{-4}\,\mathrm{s}=200\,\mu s \Rightarrow 2T_{\text{prop}}=400\,\mu s \]
	\[ T_{\text{trans}}=\frac{S}{5\times 10^6} \Rightarrow S=(5\times 10^6)(4\times 10^{-4})=2000\,\text{bytes}. \]
    \item \textbf{Binary Exponential Backoff:}
    \begin{itemize}
        \item After $m$-th collision, choose $k$ uniformly from $\{0, \ldots, 2^m-1\}$.
        \item Wait $k \cdot \text{slot time}$ (where slot time $\approx 512$ bit times in Ethernet).
    \end{itemize}
    \item \textbf{Efficiency ($\eta$):} As $N \to \infty$:
    \[ \eta = \frac{1}{1 + 2 \cdot \frac{T_{\text{prop}}}{T_{\text{trans}}} \cdot (e-1)} \]
\end{itemize}

\subsubsection{CSMA/CA (Collision Avoidance) - Wireless (WiFi/802.11)}
\begin{itemize}
    \item \textbf{Reasoning:} Cannot detect collisions reliably (weak received signal vs strong transmitted signal), so aim to avoid them.
    \item \textbf{Protocol:}
    \begin{itemize}
        \item If channel idle for \textbf{DIFS}, transmit entire frame.
        \item If busy, choose random backoff. \textbf{Timer counts down only when channel is idle} (freezes when busy).
        \item Receiver sends \textbf{ACK} after \textbf{SIFS} if frame received successfully (no ACK = collision).
    \end{itemize}
    \item \textbf{Virtual Carrier Sensing (RTS/CTS):} Prevents "Hidden Terminal" problem.
    \begin{itemize}
        \item Sender transmits \textbf{RTS} (Request to Send).
        \item Receiver replies \textbf{CTS} (Clear to Send).
        \item Neighbors hear RTS/CTS and defer transmission (reserve the channel).
    \end{itemize}
\end{itemize}

\subsection{Single (Logical) Link in practice}
\subsubsection{MAC Addresses}
\begin{itemize}
	\item Link-layer address unique per NIC (Network Interface Card); assigned by IEEE OUI.
	\item Used to deliver frames on same LAN; flat (non-hierarchical), portable across LANs.
	\item Usually burned-in, but can be software-set.
\end{itemize}
\subsubsection{Ethernet}
\begin{itemize}
	\item Dominant wired LAN; many speed generations; cheap NICs.
	\item \textbf{CSMA/CD (classic shared medium):}
	\begin{enumerate}
		\item Sense channel; if idle transmit, else wait.
		\item If collision detected while transmission, abort and send jam.
		\item \textbf{Binary exponential backoff:} after $m$-th collision pick $k\in\{0,\ldots,2^m-1\}$, wait $k\cdot512$ bit-times, retry.
	\end{enumerate}
\end{itemize}

\subsection{Interconnecting broadcast domains}
\begin{itemize}
	\item \textbf{Switch:}
    \begin{itemize}
        \item link-layer device that breaks broadcast domains; stores/forwards frames using MAC table.
        \item Transparent to hosts; self-learning (no manual config), entries age out (TTL).
        \item Each link is its own collision domain; enables simultaneous transmissions.
        \item Forwarding: if dest known on incoming port $\Rightarrow$ drop; else forward to port; if unknown $\Rightarrow$ flood.
		\item \textbf{Switch table:} (MAC, port, TTL). Learn from source MAC+ingress port on each frame.
    \end{itemize}
    \item \textbf{Loops:} cause broadcast storms, multiple frame copies, MAC table instability.
	\item \subsubsection{Spanning Tree Protocol (STP):} 
	\begin{itemize}
        \item \textbf{Goal:} Prevent Layer 2 loops by logically blocking redundant paths; converges to a loop-free spanning tree.
        \item \textbf{Mechanism:} Switches exchange \textbf{BPDUs} (Bridge Protocol Data Units) containing (RootID, Cost, SenderID, PortID).
        \item \textbf{Algorithm Steps:}
        \begin{itemize}
            \item \textbf{1. Elect Root Bridge:} Switch with the lowest \textbf{Bridge ID} becomes Root. (Initially, all claim to be Root).
            \item \textbf{2. Select Root Ports (RP):} Each \emph{non-root} switch selects \textbf{one} port with the best path to the Root.
            \item \textbf{3. Select Designated Ports (DP):} Each \emph{link/segment} selects \textbf{one} port to forward traffic (from the switch with the best path to Root).
            \item \textbf{4. Block Others:} All non-RP and non-DP ports enter \textbf{Blocking} state (receive BPDUs, but drop data).
        \end{itemize}
        \item \textbf{Tie-Breakers (Lowest wins):}
        \begin{enumerate}
            \item Lowest \textbf{Root Path Cost}.
            \item Lowest \textbf{Sender Bridge ID} (upstream switch).
            \item Lowest \textbf{Sender Port ID}.
        \end{enumerate}
    \end{itemize}
\end{itemize}

\subsection{Error Detection \& Correction}
\begin{itemize}
    \item \textbf{Key Relation:} To detect $d$ errors, need Hamming Distance $d_{min} \ge d+1$. To correct $c$ errors, need $d_{min} \ge 2c+1$.
    \item \textbf{Repetition (3x):} Send bit 3 times. Majority vote corrects 1 error. High overhead.
    \item \textbf{1D Parity:} Add 1 bit (odd/even). Detects odd \# of errors. $d_{min}=2$.
    \item \textbf{2D Parity:} Matrix of bits. Row+Col parities.
    \begin{itemize}
        \item Detects up to 3 errors (guaranteed).
        \item Fails to detect 4 errors if they form a rectangle.
        \item Corrects 1 error (intersection of bad row/col).
    \end{itemize}
    \item \textbf{Internet Checksum:} Sum 16-bit words (1's complement), append inverse. Used in IP/TCP. Detects most errors.
\end{itemize}

\section{"Layer 2.5" - ARP}
ARP is generally considered to be part of Layer 2 (Data Link Layer), but it works directly to support Layer 3 (Network Layer).
\begin{itemize}
    \item Why it's Layer 2: ARP messages are encapsulated directly into Ethernet frames. They do not have an IP header (Layer 3) and they cannot be routed outside of your local network (LAN).
    \item Why it feels like Layer 3: Its entire purpose is to handle Layer 3 information (IP addresses). It is the mechanism that allows Layer 3 to actually function on physical hardware.
\end{itemize}
\subsection{Address Resolution Protocol (ARP)}
\begin{itemize}
	\item Translates between MAC addresses and IP addresses on a LAN.
	\item Needed because Layer 3 provides the IP, but frames on Layer 2 need a destination MAC.
	\item To send outside the LAN, the packet uses the IP of the destination but the MAC of the next-hop router; routers then forward using their own L2 on each hop.
	\item Hosts use DHCP and routing to know whether the destination IP is on the same LAN or remote.
	\item Each node maintains an \textbf{ARP cache} (IP $\rightarrow$ MAC table).
	\item If the mapping is missing, send an \textbf{ARP request} (broadcast in the local broadcast domain).
	\item The target replies with an \textbf{ARP response} containing its MAC, which is stored in the cache.
	\item ARP messages stay within a single broadcast domain (not routed).
\end{itemize}
