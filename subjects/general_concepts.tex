\section{General Concepts}
\subsection{General Terms}
\begin{itemize}
    \item \textbf{Bandwidth:} max data rate of a link (bits/sec).
    \item \textbf{Goodput:} proportion of successful slots (transmission without collision).
    \item \textbf{Throughput:} proportion of slots with a transmission (success or not).
    \item \textbf{Propagation time:} time for a signal to travel between two nodes.
\end{itemize}

\subsection{Mathematical Foundations}
\subsubsection{Probability}
\begin{itemize}
    \item \textbf{Conditional:} $P(A\mid B)=\frac{P(A\cap B)}{P(B)}$, $P(B)>0$
    \item \textbf{Bayes:} $P(A\mid B)=\frac{P(B\mid A)P(A)}{P(B)}$
    \item \textbf{Total Prob. Law:} $P(B)=\sum_i P(B\mid A_i)P(A_i)$ (partition $\{A_i\}$)
    \item \textbf{Expected Value:} $\mathbb{E}[X]=\sum_x xP(X=x)$
    \item \textbf{Memoryless:} $P(X>s+t\mid X>t)=P(X>s)$ (Geo/Exp).
\end{itemize}

\begin{itemize}
    \item \textbf{Poisson Process:} events occur at rate $\lambda$; inter-arrival times $\sim \mathrm{Exp}(\lambda)$.
    \item In a Poisson process with rate $\lambda$, the probability of $k$ events in time $T$ is $P_k(t=T)=\frac{(\lambda T)^k}{k!}e^{-\lambda T}$.
\end{itemize}

\subsection{The OSI Model (5-Layer model)}
\begin{itemize}
    \item \textbf{Physical:} Move bits on a link (signals, timing, encoding).
    \item \textbf{Link:} Frames on local network, MAC addressing, error detection.
    \item \textbf{Network:} Packet delivery across networks, routing (IP).
    \item \textbf{Transport:} End-to-end process communication, reliability/flow control (TCP/UDP).
    \item \textbf{Application:} End-user services/protocols (DNS, DHCP, HTTP, FTP etc.).
\end{itemize}
